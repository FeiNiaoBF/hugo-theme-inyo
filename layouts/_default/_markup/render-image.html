{{- /* render-image.html — 图片渲染 hook Page Bundle 资源: .Page.Resources.Get 解析 → RelPermalink + width/height 静态资源 (/img/...): 保持现有子路径处理 外部 URL (http/https///data): 不处理 title: Markdown 写法 .Title 优先 loading/decoding: 首图 eager，后续 lazy */ -}} {{- $lazy := "loading=\"lazy\" decoding=\"async\"" -}} {{- if not (.Page.Scratch.Get "inyo-first-img") -}} {{- $lazy = "decoding=\"async\"" -}} {{- .Page.Scratch.Set "inyo-first-img" true -}} {{- end -}} {{- $destination := .Destination -}} {{- $lowerDestination := lower $destination -}} {{- $isExternal := or (hasPrefix $lowerDestination "http://") (hasPrefix $lowerDestination "https://") (hasPrefix $lowerDestination "//") (hasPrefix $lowerDestination "data:") -}} {{- $src := $destination -}} {{- $width := "" -}} {{- $height := "" -}} {{- if not $isExternal -}} {{- /* Page Bundle 资源优先（按 basename 匹配） */ -}} {{- $res := .Page.Resources.Get (path.Base $destination) -}} {{- with $res -}} {{- $src = .RelPermalink -}} {{- if reflect.IsImageResource $res -}} {{- $width = printf " width=\"%d\"" .Width -}} {{- $height = printf " height=\"%d\"" .Height -}} {{- end -}} {{- else -}} {{- /* 静态资源回退 */ -}} {{- $assetPath := $destination -}} {{- $isRootAsset := or (hasPrefix $destination "/") (hasPrefix $destination "img/") (hasPrefix $destination "fonts/") -}} {{- if $isRootAsset -}} {{- if not (hasPrefix $assetPath "/") }}{{ $assetPath = printf "/%s" $assetPath }}{{ end -}} {{- $src = urls.JoinPath (urls.Parse site.BaseURL).Path (strings.TrimPrefix "/" $assetPath) -}} {{- end -}} {{- end -}} {{- end -}} {{- $title := "" -}} {{- with .Title }}{{ $title = printf " title=\"%s\"" (. | htmlEscape) }}{{ end -}} {{- $img := printf "" ($src | safeURL) (.Text | plainify | htmlEscape) $width $height $title $lazy -}} {{- $img | safeHTML -}}
